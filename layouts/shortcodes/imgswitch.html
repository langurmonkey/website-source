" " }}{{ $labels = $labels | append . }}{{ end -}} {{- end -}} {{- if ne (len $labels) (len $images) -}} {{- errorf "imgswitch: %d images but %d labels (shortcode at %s)" (len $images) (len $labels) .Position -}} {{- end -}} {{- $links := slice -}} {{- range split (.Get "links") "," -}} {{- $links = $links | append (trim . " ") -}} {{- end -}} {{- if gt (len $links) (len $images) -}} {{- warnf "imgswitch: %d images but %d links, extra links ignored (shortcode at %s)" (len $images) (len $links) .Position -}} {{- end -}} {{- $active := .Get "active" | default "0" | int -}} {{- if or (lt $active 0) (ge $active (len $images)) -}} {{- errorf "imgswitch: active=%d out of range, %d images (shortcode at %s)" $active (len $images) .Position -}} {{- end -}} {{- $width := .Get "width" | default "100%" -}} {{- $loading := .Get "loading" | default "lazy" -}} {{- $class := .Get "class" | default "fig-center" -}} {{- $alt := .Get "alt" -}} {{- if not $alt }}{{ $alt = .Get "caption" | plainify }}{{ end -}} {{- if not (.Page.Scratch.Get "imgsw") -}} {{- .Page.Scratch.Set "imgsw" true }} {{- end -}}
